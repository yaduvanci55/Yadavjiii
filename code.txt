async function fetchCode(){
  const t = Date.now() + Math.floor(Math.random()*99999);
  try {
    const target = encodeURIComponent('https://codes.asmultiverse.in/api/code?t=' + t);
    const res = await fetch(`https://api.cors.lol/?url=${target}`, { cache: "no-store" });
    
    if (!res.ok) return null;
    const text = await res.text();
    const m = text.match(/ASM-[A-F0-9]+/);
    return m ? m[0] : null;
  } catch(e) {
    return null;
  }
}
