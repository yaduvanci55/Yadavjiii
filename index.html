async function fetchCode(){ const t = Date.now() + Math.floor(Math.random()*99999); try { const res = await fetch(`/api/code?t=${t}`, { cache: "no-store", signal: AbortSignal.timeout(10000) // 10 second tak wait karega }); if (!res.ok) return null; const text = await res.text(); const m = text.match(/ASM-[A-F0-9]+/); return m ? m[0] : null; } catch(e) { return null; } }
